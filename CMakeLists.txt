cmake_minimum_required(VERSION 3.20)
project(PulseMix VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(pulsemix_dsp engine/src/dsp.cpp)
target_include_directories(pulsemix_dsp PUBLIC engine/include)
target_compile_options(pulsemix_dsp PRIVATE -Wall -Wextra -Wpedantic)

if(EMSCRIPTEN)
  add_executable(dsp engine/src/wasm.cpp)
  target_link_libraries(dsp PRIVATE pulsemix_dsp)
  target_link_options(dsp PRIVATE
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,worker"
    "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=0" "-sINITIAL_MEMORY=16777216"
    "-sEXPORTED_FUNCTIONS=['_dsp_create','_dsp_destroy','_dsp_left','_dsp_right','_dsp_set','_dsp_reset','_dsp_process']"
    "-sEXPORTED_RUNTIME_METHODS=['HEAPF32']")
else()
  option(PULSEMIX_BUILD_CLI "Build file processor (requires libsndfile)" ON)
  option(PULSEMIX_BUILD_LEGACY "Build the original GTK prototype" OFF)
  if(PULSEMIX_BUILD_CLI OR PULSEMIX_BUILD_LEGACY)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
  endif()
  if(PULSEMIX_BUILD_CLI)
    add_executable(pulsemix engine/src/cli.cpp)
    target_link_libraries(pulsemix PRIVATE pulsemix_dsp PkgConfig::SNDFILE)
    target_compile_options(pulsemix PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS pulsemix RUNTIME DESTINATION bin)
  endif()
  if(PULSEMIX_BUILD_LEGACY)
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
    pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
    add_executable(pulsemix-legacy pulsemix.cpp)
    target_link_libraries(pulsemix-legacy PRIVATE PkgConfig::GTK PkgConfig::PORTAUDIO PkgConfig::SNDFILE)
  endif()
  include(CTest)
  if(BUILD_TESTING)
    add_executable(dsp_tests engine/tests/dsp_tests.cpp)
    target_link_libraries(dsp_tests PRIVATE pulsemix_dsp)
    add_test(NAME dsp_invariants COMMAND dsp_tests)
  endif()
endif()
